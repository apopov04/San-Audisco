cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
 set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum supported macOS version")
endif()
project(AudioBridge VERSION 0.2.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BRIDGE_SANITIZE "Enable address/undefined sanitizers" OFF)
if(BRIDGE_SANITIZE AND NOT MSVC)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
find_path(SODIUM_INCLUDE_DIR sodium.h HINTS ${SODIUM_ROOT}/include)
find_library(SODIUM_LIBRARY NAMES sodium libsodium HINTS ${SODIUM_ROOT}/lib)
if(NOT SODIUM_INCLUDE_DIR OR NOT SODIUM_LIBRARY)
 message(FATAL_ERROR "Set SODIUM_ROOT to a libsodium installation prefix")
endif()
find_path(OPUS_INCLUDE_DIR opus/opus.h HINTS ${SODIUM_ROOT}/include)
find_library(OPUS_LIBRARY NAMES opus HINTS ${SODIUM_ROOT}/lib)
if(NOT OPUS_INCLUDE_DIR OR NOT OPUS_LIBRARY)
 message(FATAL_ERROR "Set SODIUM_ROOT to a prefix containing libopus")
endif()
add_library(bridge_core src/engine.cpp src/transport.cpp src/credentials.cpp)
target_include_directories(bridge_core PUBLIC include PRIVATE ${SODIUM_INCLUDE_DIR} ${OPUS_INCLUDE_DIR})
target_link_libraries(bridge_core PUBLIC Threads::Threads ${SODIUM_LIBRARY} ${OPUS_LIBRARY})
if(WIN32)
 target_link_libraries(bridge_core PUBLIC ws2_32 shell32 crypt32 ole32)
 target_compile_definitions(bridge_core PRIVATE SODIUM_STATIC)
 add_library(bridge_audio src/audio_windows.cpp)
 target_link_libraries(bridge_audio PUBLIC ole32 uuid avrt propsys)
elseif(APPLE)
 target_link_libraries(bridge_core PUBLIC "-framework Security" "-framework CoreFoundation")
 add_library(bridge_audio src/audio_mac.cpp)
 target_link_libraries(bridge_audio PUBLIC "-framework CoreAudio" "-framework AudioToolbox" "-framework CoreFoundation")
endif()
target_include_directories(bridge_audio PUBLIC include)
add_executable(audiobridge apps/main.cpp)
target_link_libraries(audiobridge PRIVATE bridge_core bridge_audio)
enable_testing()
add_executable(engine_test tests/engine_test.cpp)
target_link_libraries(engine_test PRIVATE bridge_core)
add_test(NAME engine COMMAND engine_test)
add_executable(transport_test tests/transport_test.cpp)
target_link_libraries(transport_test PRIVATE bridge_core)
add_test(NAME transport COMMAND transport_test)
install(TARGETS audiobridge RUNTIME DESTINATION bin)
add_executable(integration_test tests/integration_test.cpp)
target_link_libraries(integration_test PRIVATE bridge_core)
add_test(NAME integration COMMAND integration_test)
if(WIN32)
 enable_language(RC)
 add_executable(AudioBridgeApp WIN32 apps/windows/main.cpp apps/windows/app.rc)
 target_include_directories(AudioBridgeApp PRIVATE apps/windows)
 if(MSVC)
  target_link_options(AudioBridgeApp PRIVATE /MANIFEST:NO)
 endif()
 target_include_directories(AudioBridgeApp PRIVATE third_party/json/include)
 target_link_libraries(AudioBridgeApp PRIVATE comctl32 comdlg32 shell32 user32 gdi32 advapi32)
 if(MINGW)
 target_link_options(AudioBridgeApp PRIVATE -municode)
 endif()
endif()

option(BRIDGE_BUILD_ROOMS "Build native room agent and pairing tests" ON)
if(BRIDGE_BUILD_ROOMS)
 find_path(BORINGSSL_INCLUDE_DIR openssl/curve25519.h HINTS ${BORINGSSL_ROOT}/include NO_DEFAULT_PATH)
 find_library(BORINGSSL_CRYPTO NAMES crypto libcrypto HINTS ${BORINGSSL_ROOT}/lib NO_DEFAULT_PATH)
 if(NOT BORINGSSL_INCLUDE_DIR OR NOT BORINGSSL_CRYPTO)
  message(FATAL_ERROR "Build pinned room crypto with scripts/bootstrap-room-crypto.py and set BORINGSSL_ROOT, or disable BRIDGE_BUILD_ROOMS for engine-only builds")
 endif()
 add_library(bridge_rooms src/room.cpp)
 target_include_directories(bridge_rooms PUBLIC include PRIVATE third_party/json/include ${BORINGSSL_INCLUDE_DIR} ${SODIUM_INCLUDE_DIR})
 target_link_libraries(bridge_rooms PUBLIC bridge_core ${BORINGSSL_CRYPTO})
 if(WIN32)
  target_compile_definitions(bridge_rooms PRIVATE SODIUM_STATIC)
  target_link_libraries(bridge_rooms PUBLIC bcrypt)
  if(MINGW)
   target_link_libraries(bridge_rooms PUBLIC winpthread)
  endif()
 endif()
 add_executable(bridge-agent apps/agent.cpp src/agent_platform.cpp)
 target_include_directories(bridge-agent PRIVATE third_party/json/include ${SODIUM_INCLUDE_DIR})
 target_link_libraries(bridge-agent PRIVATE bridge_rooms bridge_audio)
 if(APPLE)
  enable_language(OBJCXX)
  target_sources(bridge-agent PRIVATE src/agent_permission.mm)
  target_link_libraries(bridge-agent PRIVATE "-framework AVFoundation")
 elseif(WIN32)
  target_compile_definitions(bridge-agent PRIVATE SODIUM_STATIC)
  target_link_libraries(bridge-agent PRIVATE advapi32)
 endif()
 add_executable(room_test tests/room_test.cpp)
 target_include_directories(room_test PRIVATE third_party/json/include ${SODIUM_INCLUDE_DIR})
 target_link_libraries(room_test PRIVATE bridge_rooms)
 add_test(NAME room_pairing COMMAND room_test)
 set_tests_properties(room_pairing PROPERTIES TIMEOUT 90)

 target_include_directories(room_wire_test PRIVATE third_party/json/include ${SODIUM_INCLUDE_DIR})
 target_link_libraries(room_wire_test PRIVATE bridge_rooms)

 set_tests_properties(room_wire PROPERTIES TIMEOUT 90)
 add_executable(agent_test_runtime apps/agent.cpp src/agent_platform.cpp tests/agent_audio_inventory.cpp)
 target_compile_definitions(agent_test_runtime PRIVATE BRIDGE_AGENT_FAKE)
 target_include_directories(agent_test_runtime PRIVATE third_party/json/include ${SODIUM_INCLUDE_DIR})
 target_link_libraries(agent_test_runtime PRIVATE bridge_rooms)
 if(WIN32)
  target_compile_definitions(agent_test_runtime PRIVATE SODIUM_STATIC)
  target_link_libraries(agent_test_runtime PRIVATE advapi32)
 endif()
 add_executable(agent_child_test tests/agent_child_test.cpp src/agent_platform.cpp)
 target_compile_definitions(agent_child_test PRIVATE BRIDGE_AGENT_FAKE)
 target_include_directories(agent_child_test PRIVATE include ${SODIUM_INCLUDE_DIR})
 target_link_libraries(agent_child_test PRIVATE ${SODIUM_LIBRARY} Threads::Threads)
 if(WIN32)
  target_compile_definitions(agent_child_test PRIVATE SODIUM_STATIC)
  target_link_libraries(agent_child_test PRIVATE shell32 ole32 advapi32 uuid)
 endif()
 add_test(NAME agent_child COMMAND agent_child_test)
 set_tests_properties(agent_child PROPERTIES TIMEOUT 30)
 install(TARGETS bridge-agent RUNTIME DESTINATION bin)
endif()

cmake_minimum_required(VERSION 3.20)
if(CMAKE_HOST_APPLE AND NOT DEFINED CMAKE_OSX_DEPLOYMENT_TARGET)
 set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum supported macOS version")
endif()
project(AudioBridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(BRIDGE_SANITIZE "Enable address/undefined sanitizers" OFF)
if(BRIDGE_SANITIZE AND NOT MSVC)
 add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
 add_link_options(-fsanitize=address,undefined)
endif()
find_package(Threads REQUIRED)
find_path(SODIUM_INCLUDE_DIR sodium.h HINTS ${SODIUM_ROOT}/include)
find_library(SODIUM_LIBRARY NAMES sodium libsodium HINTS ${SODIUM_ROOT}/lib)
if(NOT SODIUM_INCLUDE_DIR OR NOT SODIUM_LIBRARY)
 message(FATAL_ERROR "Set SODIUM_ROOT to a libsodium installation prefix")
endif()
find_path(OPUS_INCLUDE_DIR opus/opus.h HINTS ${SODIUM_ROOT}/include)
find_library(OPUS_LIBRARY NAMES opus HINTS ${SODIUM_ROOT}/lib)
if(NOT OPUS_INCLUDE_DIR OR NOT OPUS_LIBRARY)
 message(FATAL_ERROR "Set SODIUM_ROOT to a prefix containing libopus")
endif()
add_library(bridge_core src/engine.cpp src/transport.cpp src/credentials.cpp)
target_include_directories(bridge_core PUBLIC include PRIVATE ${SODIUM_INCLUDE_DIR} ${OPUS_INCLUDE_DIR})
target_link_libraries(bridge_core PUBLIC Threads::Threads ${SODIUM_LIBRARY} ${OPUS_LIBRARY})
if(WIN32)
 target_link_libraries(bridge_core PUBLIC ws2_32 shell32 crypt32 ole32)
 target_compile_definitions(bridge_core PRIVATE SODIUM_STATIC)
 add_library(bridge_audio src/audio_windows.cpp)
 target_link_libraries(bridge_audio PUBLIC ole32 uuid avrt propsys)
elseif(APPLE)
 target_link_libraries(bridge_core PUBLIC "-framework Security" "-framework CoreFoundation")
 add_library(bridge_audio src/audio_mac.cpp)
 target_link_libraries(bridge_audio PUBLIC "-framework CoreAudio" "-framework AudioToolbox" "-framework CoreFoundation")
endif()
target_include_directories(bridge_audio PUBLIC include)
add_executable(audiobridge apps/main.cpp)
target_link_libraries(audiobridge PRIVATE bridge_core bridge_audio)
enable_testing()
add_executable(engine_test tests/engine_test.cpp)
target_link_libraries(engine_test PRIVATE bridge_core)
add_test(NAME engine COMMAND engine_test)
add_executable(transport_test tests/transport_test.cpp)
target_link_libraries(transport_test PRIVATE bridge_core)
add_test(NAME transport COMMAND transport_test)
install(TARGETS audiobridge RUNTIME DESTINATION bin)
add_executable(integration_test tests/integration_test.cpp)
target_link_libraries(integration_test PRIVATE bridge_core)
add_test(NAME integration COMMAND integration_test)
if(WIN32)
 add_executable(AudioBridgeApp WIN32 apps/windows/main.cpp)
 target_link_libraries(AudioBridgeApp PRIVATE comctl32 comdlg32 shell32 user32 gdi32)
 if(MINGW)
 target_link_options(AudioBridgeApp PRIVATE -municode)
 endif()
endif()

add_executable(adaptive_ring_test tests/adaptive_ring_test.cpp)
target_link_libraries(adaptive_ring_test PRIVATE bridge_core)
add_test(NAME adaptive_ring COMMAND adaptive_ring_test)
add_executable(bridge_benchmark tests/benchmark.cpp)
target_link_libraries(bridge_benchmark PRIVATE bridge_core)
add_executable(runtime_fake apps/main.cpp tests/audio_fake.cpp)
target_link_libraries(runtime_fake PRIVATE bridge_core)
find_package(Python3 COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
 add_test(NAME runtime COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_test.py $<TARGET_FILE:runtime_fake>)
 set_tests_properties(runtime PROPERTIES TIMEOUT 120)
 add_test(NAME credentials_cli COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/credentials_cli_test.py $<TARGET_FILE:runtime_fake>)
 set_tests_properties(credentials_cli PROPERTIES TIMEOUT 30)
 add_test(NAME optional_routes COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/optional_routes_test.py $<TARGET_FILE:runtime_fake>)
 set_tests_properties(optional_routes PROPERTIES TIMEOUT 60)
endif()
add_executable(control_test tests/control_test.cpp)
target_link_libraries(control_test PRIVATE bridge_core)
add_test(NAME control COMMAND control_test)
if(APPLE)
 add_executable(mac_audio_lifecycle_test tests/mac_audio_lifecycle_test.cpp)
 target_include_directories(mac_audio_lifecycle_test PRIVATE include)
 target_compile_options(mac_audio_lifecycle_test PRIVATE -fno-access-control)
 target_link_libraries(mac_audio_lifecycle_test PRIVATE "-framework AudioUnit" "-framework CoreAudio" "-framework CoreFoundation")
 add_test(NAME mac_audio_lifecycle COMMAND mac_audio_lifecycle_test)
 set_tests_properties(mac_audio_lifecycle PROPERTIES SKIP_RETURN_CODE 77)
endif()

add_executable(credentials_test tests/credentials_test.cpp)
target_link_libraries(credentials_test PRIVATE bridge_core)
add_test(NAME credentials COMMAND credentials_test)
add_executable(listening_mixer_test tests/listening_mixer_test.cpp)
target_link_libraries(listening_mixer_test PRIVATE bridge_core)
add_test(NAME listening_mixer COMMAND listening_mixer_test)

if(WIN32)
 add_executable(windows_audio_smoke tests/windows_audio_smoke.cpp)
 target_link_libraries(windows_audio_smoke PRIVATE bridge_audio)
 add_test(NAME windows_audio_smoke COMMAND windows_audio_smoke)
 set_tests_properties(windows_audio_smoke PROPERTIES SKIP_RETURN_CODE 77)
endif()

if(WIN32)
 add_executable(windows_audio_recovery_test tests/windows_audio_recovery_test.cpp)
 target_include_directories(windows_audio_recovery_test PRIVATE include)
 target_link_libraries(windows_audio_recovery_test PRIVATE ole32 uuid avrt propsys)
 add_test(NAME windows_audio_recovery COMMAND windows_audio_recovery_test)
 set_tests_properties(windows_audio_recovery PROPERTIES SKIP_RETURN_CODE 77)
endif()
